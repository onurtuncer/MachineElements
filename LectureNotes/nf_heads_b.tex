\newcommand{\shank}[3]{%
  \draw[steel] (#1-0.5,0) rectangle (#1+0.5,-#2);
  \foreach \k in {0,1,...,9}{%
    \pgfmathsetmacro\yy{-#2*(#3+(1-#3)*(0.07+0.098*\k))}
    \ifdim\yy pt>-#2 pt
      \draw[blue!55!black,thin] (#1-0.5,\yy) -- (#1+0.5,\yy-0.085);
    \fi}
  \draw[blue!55!black,thin,dashed] (#1-0.5,-#2*#3) -- (#1+0.5,-#2*#3);
}
